% \diagraminput[maxheight-mm]{name} — scales the TikZ fragment to fit inside
% both the text width and the given height, so a tall diagram cannot push a
% page over. Centred, because a height-capped diagram is narrower than the
% measure.
\newcommand{\diagraminput}[2][95]{%
  \par\vspace{2mm}%
  \IfFileExists{images/diagrams/#2.tex}{%
    \noindent\makebox[\textwidth][c]{%
      \adjustbox{max width=\textwidth,max totalheight=#1mm}{%
        \input{images/diagrams/#2.tex}}}%
  }{%
    \brochurefigure{#2}{1.0}{#1}%
  }%
  \par\vspace{2mm}%
}

% ---------------------------------------------------------------------------
% The page model
% ---------------------------------------------------------------------------
%
% Every physical text page is one snarkypage environment. Content is captured
% into a box first, and its natural height reported to the log, so the build
% tool can tell the author which pages overflow or run short. Fit is a build
% diagnostic, not something discovered by eye.

\newsavebox{\snarkypagebox}
\newsavebox{\snarkynatbox}

\newenvironment{snarkypage}[2]{%
  \def\snarkypagenum{#1}%
  \def\snarkypagelayout{#2}%
  \begin{lrbox}{\snarkynatbox}%
  \begin{minipage}[t]{\textwidth}%
  % minipage applies \@parboxrestore, which zeroes \parskip. Restore the
  % document's paragraph separation inside the page box.
  \setlength{\parskip}{\bodyparskip}%
  \setlength{\parindent}{0pt}%
}{%
  \end{minipage}%
  \end{lrbox}%
  \typeout{PAGEFIT page=\snarkypagenum\space layout=\snarkypagelayout\space natural=\the\ht\snarkynatbox\space depth=\the\dp\snarkynatbox\space avail=\the\dimexpr\textheight-\folioheight\relax}%
  \cropmarks
  \noindent\begin{minipage}[t][\dimexpr\textheight-\topskip\relax][s]{\textwidth}%
    \usebox{\snarkynatbox}%
    \vfill
    \pagefolio{\snarkypagenum}%
  \end{minipage}%
  \clearpage
}
